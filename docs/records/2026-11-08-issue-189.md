```sh
pnpm --filter server... build && PATH="/opt/homebrew/opt/openjdk@21/bin:$PATH" PLAYWRIGHT_MODULE=/path/to/playwright/index.mjs pnpm exec firebase emulators:exec --project demo-kadai-hisho --only auth,firestore "node packages/web/tests/run-plans-browser.mjs --capture-landing-shots"
```
